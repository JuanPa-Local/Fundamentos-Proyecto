-- -------------------------------------------------------------------------
-- PASO 1: LIMPIAR TABLAS ANTIGUAS
-- -------------------------------------------------------------------------
DROP TABLE IF EXISTS order_items CASCADE;
DROP TABLE IF EXISTS orders CASCADE;
DROP TABLE IF EXISTS reviews CASCADE;
DROP TABLE IF EXISTS favorites CASCADE;
DROP TABLE IF EXISTS books CASCADE;
DROP TABLE IF EXISTS users CASCADE;

-- -------------------------------------------------------------------------
-- PASO 2: CREAR TABLAS (Misma estructura que usa Spring Boot)
-- -------------------------------------------------------------------------

CREATE TABLE users (
    id UUID PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL,
    created_at TIMESTAMP NOT NULL,
    active BOOLEAN NOT NULL,
    phone VARCHAR(255),
    address VARCHAR(255),
    avatar_url VARCHAR(255)
);

CREATE TABLE books (
    id UUID PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    author VARCHAR(255) NOT NULL,
    isbn VARCHAR(255),
    description TEXT,
    category VARCHAR(255),
    file_path VARCHAR(255),
    cover_url VARCHAR(255),
    status VARCHAR(30) DEFAULT 'PENDIENTE',
    seller_email VARCHAR(255),
    rejection_reason TEXT,
    price DOUBLE PRECISION DEFAULT 0.0,
    created_at TIMESTAMP NOT NULL,
    download_count INTEGER DEFAULT 0,
    average_rating DOUBLE PRECISION DEFAULT 0.0,
    review_count INTEGER DEFAULT 0
);

CREATE TABLE orders (
    id UUID PRIMARY KEY,
    user_id UUID,
    book_id UUID,
    status VARCHAR(30) DEFAULT 'COMPLETED',
    total_price DOUBLE PRECISION DEFAULT 0.0,
    ordered_at TIMESTAMP NOT NULL,
    CONSTRAINT fk_order_user FOREIGN KEY (user_id) REFERENCES users (id),
    CONSTRAINT fk_order_book FOREIGN KEY (book_id) REFERENCES books (id)
);

CREATE TABLE reviews (
    id UUID PRIMARY KEY,
    user_id UUID NOT NULL,
    book_id UUID NOT NULL,
    rating INTEGER NOT NULL,
    comment TEXT,
    created_at TIMESTAMP NOT NULL,
    status VARCHAR(30) DEFAULT 'ACTIVA',
    CONSTRAINT fk_review_user FOREIGN KEY (user_id) REFERENCES users (id),
    CONSTRAINT fk_review_book FOREIGN KEY (book_id) REFERENCES books (id),
    CONSTRAINT uk_review_user_book UNIQUE (user_id, book_id)
);

CREATE TABLE favorites (
    id UUID PRIMARY KEY,
    user_id UUID NOT NULL,
    book_id UUID NOT NULL,
    added_at TIMESTAMP NOT NULL,
    CONSTRAINT fk_favorite_user FOREIGN KEY (user_id) REFERENCES users (id),
    CONSTRAINT fk_favorite_book FOREIGN KEY (book_id) REFERENCES books (id),
    CONSTRAINT uk_favorite_user_book UNIQUE (user_id, book_id)
);
